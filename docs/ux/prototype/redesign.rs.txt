/// The proposed type ramp: four sizes only.
fn ty_title(p: Paint) -> TextStyle {
    TextStyle {
        size: 17.0,
        line_height: 22.0,
        ..p.k.ty.ui_medium
    }
}
fn ty_body(p: Paint) -> TextStyle {
    TextStyle {
        size: 14.0,
        line_height: 20.0,
        ..p.k.ty.ui
    }
}
fn ty_label(p: Paint) -> TextStyle {
    TextStyle {
        size: 12.0,
        line_height: 16.0,
        ..p.k.ty.small_medium
    }
}
fn ty_meta(p: Paint) -> TextStyle {
    TextStyle {
        size: 11.5,
        line_height: 15.0,
        ..p.k.ty.small
    }
}

/// The assistant's own hue (global states: listening, the desk frame).
fn agent(p: Paint) -> (Color, Color) {
    palette::frame(p.k.dark)
}

/// The proposed task palette: six hues that stay clear of the semantic
/// green (done) and red (failed), and of the assistant's periwinkle.
fn task(p: Paint, index: usize) -> Color {
    let dark = p.k.dark;
    let hues = [
        (205.0, 0.15),
        (60.0, 0.16),
        (170.0, 0.12),
        (340.0, 0.17),
        (95.0, 0.14),
        (25.0, 0.10),
    ];
    let (h, c) = hues[index % hues.len()];
    if dark {
        oklch(0.76, c, h)
    } else {
        oklch(0.6, c, h)
    }
}

fn done(p: Paint) -> Color {
    if p.k.dark {
        oklch(0.78, 0.16, 150.0)
    } else {
        oklch(0.58, 0.15, 150.0)
    }
}
fn failed(p: Paint) -> Color {
    if p.k.dark {
        oklch(0.72, 0.17, 25.0)
    } else {
        oklch(0.56, 0.19, 25.0)
    }
}
fn needs(p: Paint) -> Color {
    if p.k.dark {
        oklch(0.84, 0.14, 85.0)
    } else {
        oklch(0.7, 0.15, 75.0)
    }
}

fn shadow(blur: f32, y: f32, alpha: f32) -> Shadow {
    Shadow {
        x: 0.0,
        y,
        blur,
        spread: 0.0,
        color: Color::BLACK.alpha(alpha),
        inset: false,
    }
}

/// The one card material every bubble uses.
fn material(p: Paint, e: El) -> El {
    let c = &p.k.c;
    e.radius(16.0)
        .fill(c.bg_base.alpha(1.0))
        .border(p.hair, c.border_base)
        .box_shadow(shadow(28.0, 10.0, if p.k.dark { 0.45 } else { 0.16 }))
}

fn dot(colour: Color, size: f32) -> El {
    el().size(size, size).radius(size / 2.0).fill(colour)
}

/// A pill button: `primary` is the one accent action of a card.
fn button(p: Paint, label: &str, primary: bool) -> El {
    let c = &p.k.c;
    let b = row()
        .h(28.0)
        .px(12.0)
        .center_y()
        .radius(14.0)
        .child(text(label.to_owned()).font(ty_label(p)).color(if primary {
            c.bg_base
        } else {
            p.text()
        }));
    if primary {
        b.fill(c.bg_inverse)
    } else {
        b.fill(c.surface_raised_base).border(p.hair, c.border_base)
    }
}

/// A keycap hint, "Hold ⌘".
fn key_hint(p: Paint, words: &str) -> El {
    text(words.to_owned()).font(ty_meta(p)).color(p.faint())
}

/// A state chip: glyph + word, never colour alone.
fn chip(p: Paint, glyph: &'static Icon, words: &str, colour: Color) -> El {
    row()
        .h(20.0)
        .px(7.0)
        .gap(4.0)
        .center_y()
        .radius(10.0)
        .fill(colour.alpha(0.16))
        .child(icon(glyph, 11.0).color(colour))
        .child(text(words.to_owned()).font(ty_meta(p)).color(colour))
}

/// The card that hangs under the island: the one bubble anatomy.
///
/// header: glyph · who/what · state chip (right)
/// body: content (live words, reasoning, answer, question)
/// footer: ≤ 1 primary + ≤ 2 secondary actions, the voice hint left
struct Card {
    lead: El,
    title: String,
    chip: Option<El>,
    body: Vec<El>,
    hint: Option<String>,
    actions: Vec<El>,
    width: f32,
}

fn card(p: Paint, card: Card) -> El {
    let header = row()
        .w_full()
        .gap(8.0)
        .center_y()
        .child(card.lead)
        .child(
            text(card.title)
                .font(ty_label(p))
                .color(p.muted())
                .truncate()
                .grow(1.0)
                .min_w(0.0),
        )
        .children(card.chip);
    let footer = (card.hint.is_some() || !card.actions.is_empty()).then(|| {
        row()
            .w_full()
            .mt(4.0)
            .gap(8.0)
            .center_y()
            .child(
                el().grow(1.0)
                    .children(card.hint.map(|h| key_hint(p, &h))),
            )
            .children(card.actions)
    });
    material(
        p,
        col()
            .w(card.width)
            .padding(16.0)
            .gap(10.0)
            .child(header)
            .children(card.body)
            .children(footer),
    )
}

fn body(p: Paint, words: &str) -> El {
    text(words.to_owned()).font(ty_body(p)).color(p.text()).wrap()
}

/// One reasoning line: a small label column and the words.
fn thought(p: Paint, label: &str, words: &str, state: Step) -> El {
    let c = &p.k.c;
    let mark = match state {
        Step::Done => icon(&CHECK, 12.0).color(done(p)),
        Step::Now => el().size(12.0, 12.0).center().child(dot(agent(p).0, 7.0)),
        Step::Next => el()
            .size(12.0, 12.0)
            .center()
            .child(el().size(7.0, 7.0).radius(3.5).border(1.2, c.border_strong)),
    };
    row()
        .w_full()
        .gap(8.0)
        .items_start()
        .child(el().pt(4.0).child(mark))
        .child(
            text(label.to_owned())
                .w(76.0)
                .font(ty_meta(p))
                .color(p.faint())
                .pt(2.0),
        )
        .child(
            text(words.to_owned())
                .font(ty_body(p))
                .color(match state {
                    Step::Next => p.faint(),
                    Step::Done => p.muted(),
                    Step::Now => p.text(),
                })
                .wrap()
                .grow(1.0)
                .min_w(0.0),
        )
}

#[derive(Clone, Copy)]
enum Step {
    Done,
    Now,
    Next,
}

fn under_island(e: El, width: f32) -> El {
    el().absolute((W - width) / 2.0, 70.0).w(width).child(e)
}

/// Tier 2: a window outline in its task's colour, with its chip.
fn outline(p: Paint, (x, y, w, h): (f32, f32, f32, f32), colour: Color, words: &str, step: &str, dashed: bool) -> El {

#[derive(Clone, Copy, Debug)]
pub(crate) enum Scene {
    Overview,
    Listening,
    Foreground,
    Stage,
    Results,
    IslandStates,
    Frames,
    Anatomy,
    Minis,
}

impl Scene {
    pub(crate) fn size(self) -> (f32, f32) {
        match self {
            Scene::IslandStates => (1000.0, 1480.0),
            Scene::Frames => (1320.0, 470.0),
            Scene::Anatomy => (900.0, 420.0),
            Scene::Minis => (820.0, 350.0),
            _ => (W, H),
        }
    }
}

pub(crate) fn view(scene: &Scene, cx: &mut Cx<Scene>) -> El {
    let p = Paint::new(cx.system_dark(), cx.env().scale);
    match scene {
        Scene::Overview => overview(p),
        Scene::Listening => listening(p),
        Scene::Foreground => foreground(p),
        Scene::Stage => stage(p),
        Scene::Results => results(p),
        Scene::IslandStates => island_states(p),
        Scene::Frames => frames(p),
        Scene::Anatomy => anatomy(p),
        Scene::Minis => minis(p),
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use rune::Env;
    use rune::text::TextSystem;

    #[test]
    #[ignore = "set HEX_REDESIGN_SNAPS to a folder; needs a GPU"]
    fn redesign_snapshots() {
        let folder = std::path::PathBuf::from(std::env::var("HEX_REDESIGN_SNAPS").unwrap());
        std::fs::create_dir_all(&folder).unwrap();
        let gpu = rune::render::Gpu::try_new(rune::render::Gpu::instance(), None).unwrap();
        let scenes = [
            ("overview", Scene::Overview),
            ("listening", Scene::Listening),
            ("foreground", Scene::Foreground),
            ("stage", Scene::Stage),
            ("results", Scene::Results),
            ("island-states", Scene::IslandStates),
            ("frames", Scene::Frames),
            ("anatomy", Scene::Anatomy),
            ("minis", Scene::Minis),
        ];
        for dark in [false, true] {
            let theme = if dark { "dark" } else { "light" };
            for (name, scene) in scenes {
                let size = scene.size();
                let mut rt = crate::rune_ui::offscreen::runtime(
                    scene,
                    view,
                    size,
                    Env {
                        system_dark: dark,
                        scale: 2.0,
                    },
                    TextSystem::new(rune_theme_opencode::FONTS, rune_theme_opencode::FAMILIES),
                );
                let (w, h) = ((size.0 * 2.0) as u32, (size.1 * 2.0) as u32);
                let pixels = crate::rune_ui::offscreen::draw(&gpu, &mut rt, w, h);
                rune::render::write_png(&folder.join(format!("proposed-{name}-{theme}.png")), w, h, &pixels)
                    .unwrap();
            }
        }
    }
}
